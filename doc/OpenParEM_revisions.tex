\begin{itemize}[nosep]

   \item version 1.0.1 [9-18-2024]
      \begin{itemize}[nosep]
         \item Initial release.
      \end{itemize}

   \item version 1.0.2 [9-25-2024]
      \begin{itemize}[nosep]
         \item Fixed small memory leak in OpenParEM3D
         \item Updated installation documentation for Ubuntu 22.04.5 LTS
         \item Started change log
      \end{itemize}

   \item version 1.0.3 [10-5-2024]
      \begin{itemize}[nosep]
         \item Fixed a memory cleanup error
         \item Ubuntu 24 now supported
         \item Updated installation documentation for Ubuntu 24.04.1 LTS
      \end{itemize}

   \item version 2.0.0 [3-6-25]
      \begin{itemize}[nosep]
         \item Added antenna performance metrics.
         \item Upgraded adaptive mesh refinement with a new pre-conditioner.
      \end{itemize}

   \item version 2.1.0 [5-1-2025]
      \begin{itemize}[nosep]
         \item Re-organized the code base into a single root directory.
         \item Added a script to enable compiling from a single command and verified on multiple Linux platforms.
         \item Added a binary build requiring no package installations and verified on multiple Linux platforms.
         \item Upgraded libraries
            \begin{itemize}[nosep]
               \item Hypre from 2.30.0 to 2.32.0
               \item PETSc from 3.19.6 to 3.23.0
               \item Metis from 4.0.3 to v5.1.0-p12
               \item MFEM from 4.7 to 4.8
            \end{itemize}
      \end{itemize}

   \item version 2.1.0 [3-20-2025]
      \begin{itemize}[nosep]
         \item Updated system calls for improved timing calculations, especially when using Windows WSL2
         \item Documentation update to include support for Windows WSL2
      \end{itemize}

   \item version 2.2.0 [9-1-2026]
      \begin{itemize}[nosep]
         \item Initial release of OpenParEMg with users manual.
         \item Numerous small changes to OpenParEM3D and OpenParEM2D to support OpenParEMg
         \item Update OpenParEM\_build to support OpenParEMg
         \item Updated the regression suite to run with either OpenParEMg or by script on the FreeCAD/gmsh builds
         \item Modified OpenParEM\_build to download and verify packages rather than include the packages.  Switched to latest release in most cases.
      \end{itemize}

\end{itemize}
